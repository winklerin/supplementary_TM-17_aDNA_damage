2. Plots

data_long <- data %>% 
  mutate(sample=rownames(data)) %>%  
  mutate(sample = str_remove(sample, "^MEGAHIT-group-")) %>% 
  mutate(category=(case_when(
    str_starts(sample, "EXB") ~ "EXB",
    str_starts(sample, "TDM") ~ "TDM",
    TRUE ~ NA_character_  # for any other cases
  ))) %>% 
  pivot_longer(cols= -c(sample, category))
desired_order=paste0(rep('TDM0', times=61), c('01','02','03','04','05','06', '07', '08', '09', as.character(10:61)))
data_long$depth_cm <- as.numeric(sub("TDM0*", "", data_long$sample))
## Warning: NAs introduced by coercion
p1= data_long[data_long$name %in% c( "Total_length_gt_0_bp"  ) & data_long$category=='TDM',] %>% 
  ggplot(aes(y=depth_cm-0.5, x=value))+
   geom_segment( aes(y=depth_cm-0.5, yend=depth_cm-0.5, x=data_long[data_long$name %in% c( "Total_length_gt_0_bp"  ) & data_long$category=='EXB',]$value %>% max(), xend=value), color="grey40") +
  geom_point( color="grey40", size=1.5)+
  geom_vline(xintercept = data_long[data_long$name %in% c( "Total_length_gt_0_bp"  ) & data_long$category=='EXB',]$value %>% max(), color='darkred')+
   scale_y_reverse( breaks = seq(1, 61, 1), labels=ifelse(1:61 %% 5 == 0, 1:61, ""), limits = c(61,-3),expand = c(0, 0))+
  theme_classic(base_size = 9)+
  theme(axis.text.x = element_text(angle = 30, vjust = 0.8, hjust = 1), 
        axis.ticks.length = unit(0.1, "cm"),
        panel.grid.major = element_line(color = "grey90")) +
  ylab('Depth from top (cm)')+
  xlab('Assembly length (bp)')
